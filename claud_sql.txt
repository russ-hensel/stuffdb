SELECT
    photoshow.id,
    photoshow.name,
    photoshow.cmnt
FROM
    photoshow
    JOIN photo_in_show ON photo_in_show.photo_show_id = photoshow.id
WHERE
    photo_in_show.photo_id = a_given_photo_id
;
